```python
class QueryOptimizer:
    """查询优化器"""

    def __init__(self, storage: OptimizedSQLAlchemyEpisodeStorage):
        self.storage = storage
        self.query_cache = LRUCache(maxsize=1000, ttl=300)  # 5分钟缓存
        self.query_stats = defaultdict(int)

    async def get_episodes_optimized(
        self,
        session_key: str,
        limit: int | None = None,
        page_num: int | None = None,
        filter_expr: FilterExpr | None = None
    ) -> list[Episode]:
        """优化的对话获取"""

        # 生成缓存键
        cache_key = self._generate_cache_key(
            session_key, limit, page_num, filter_expr
        )

        # 检查缓存
        cached_result = self.query_cache.get(cache_key)
        if cached_result is not None:
            self.query_stats['cache_hits'] += 1
            return cached_result

        # 生成优化的SQL查询
        query, params = self._build_optimized_query(
            session_key, limit, page_num, filter_expr
        )

        # 执行查询
        start_time = time.time()
        async with self.storage._engine.connect() as conn:
            result = await conn.execute(query, params)
            rows = result.fetchall()

        execution_time = time.time() - start_time

        # 记录性能指标
        self.query_stats['cache_misses'] += 1
        self.query_stats['execution_times'].append(execution_time)

        # 转换为Episode对象
        episodes = self._rows_to_episodes(rows)

        # 缓存结果
        if len(episodes) > 0:
            self.query_cache[cache_key] = episodes

        return episodes

    def _build_optimized_query(
        self,
        session_key: str,
        limit: int | None,
        page_num: int | None,
        filter_expr: FilterExpr | None
    ) -> tuple[str, dict]:
        """构建优化的SQL查询"""

        # 基础查询
        query = """
            SELECT
                uid, content, session_key, created_at,
                producer_id, producer_role, produced_for_id,
                sequence_num, episode_type, content_type,
                metadata, filterable_metadata
            FROM episodes
            WHERE session_key = :session_key
        """
        params = {"session_key": session_key}

        # 添加过滤器条件
        if filter_expr:
            filter_clause, filter_params = self._build_filter_clause(filter_expr)
            query += f" AND {filter_clause}"
            params.update(filter_params)

        # 排序优化：使用时间戳索引
        query += " ORDER BY created_at_timestamp DESC"

        # 分页优化
        if limit:
            if page_num and page_num > 0:
                offset = (page_num - 1) * limit
                query += " LIMIT :limit OFFSET :offset"
                params.update({"limit": limit, "offset": offset})
            else:
                query += " LIMIT :limit"
                params["limit"] = limit

        return query, params

    def _build_filter_clause(self, filter_expr: FilterExpr) -> tuple[str, dict]:
        """构建过滤器子句"""
        if isinstance(filter_expr, FilterComparison):
            field = filter_expr.field
            op = filter_expr.op
            value = filter_expr.value

            if op == "=":
                return f"{field} = :{field}", {field: value}
            elif op == ">":
                return f"{field} > :{field}", {field: value}
            elif op == "<":
                return f"{field} < :{field}", {field: value}
            elif op == "LIKE":
                return f"{field} LIKE :{field}", {field: f"%{value}%"}
            else:
                raise ValueError(f"Unsupported operator: {op}")

        elif isinstance(filter_expr, FilterAnd):
            left_clause, left_params = self._build_filter_clause(filter_expr.left)
            right_clause, right_params = self._build_filter_clause(filter_expr.right)

            merged_params = {**left_params, **right_params}
            return f"({left_clause}) AND ({right_clause})", merged_params

        else:
            raise ValueError(f"Unsupported filter type: {type(filter_expr)}")

    def _generate_cache_key(self, *args) -> str:
        """生成缓存键"""
        import hashlib
        key_parts = [str(arg) for arg in args if arg is not None]
        return hashlib.md5("|".join(key_parts).encode()).hexdigest()

    def _rows_to_episodes(self, rows) -> list[Episode]:
        """数据库行转换为Episode对象"""
        episodes = []
        for row in rows:
            episode_dict = dict(row._mapping)
            episodes.append(Episode(**episode_dict))
        return episodes
```

```python
import asyncio
import hashlib
import time
from typing import Any, Optional, Dict
from abc import ABC, abstractmethod

class CacheBackend(ABC):
    """缓存后端抽象接口"""

    @abstractmethod
    async def get(self, key: str) -> Any:
        pass

    @abstractmethod
    async def set(self, key: str, value: Any, expire: Optional[int] = None) -> None:
        pass

    @abstractmethod
    async def delete(self, key: str) -> None:
        pass

    @abstractmethod
    async def clear(self) -> None:
        pass

class RedisCache(CacheBackend):
    """Redis缓存实现"""

    def __init__(self, redis_url: str):
        self.redis_url = redis_url
        self._redis = None

    async def initialize(self):
        import aioredis
        self._redis = await aioredis.from_url(self.redis_url)

    async def get(self, key: str) -> Any:
        if not self._redis:
            return None

        try:
            data = await self._redis.get(key)
            if data:
                import pickle
                return pickle.loads(data)
        except Exception as e:
            logging.error(f"Redis cache get error: {e}")

        return None

    async def set(self, key: str, value: Any, expire: Optional[int] = None) -> None:
        if not self._redis:
            return

        try:
            import pickle
            data = pickle.dumps(value)
            await self._redis.setex(key, expire or 3600, data)
        except Exception as e:
            logging.error(f"Redis cache set error: {e}")

    async def delete(self, key: str) -> None:
        if self._redis:
            await self._redis.delete(key)

    async def clear(self) -> None:
        if self._redis:
            await self._redis.flushdb()

class MemoryCache(CacheBackend):
    """内存缓存实现"""

    def __init__(self, max_size: int = 1000):
        self.max_size = max_size
        self._cache: Dict[str, Any] = {}
        self._access_times: Dict[str, float] = {}

    async def get(self, key: str) -> Any:
        if key in self._cache:
            self._access_times[key] = time.time()
            return self._cache[key]
        return None

    async def set(self, key: str, value: Any, expire: Optional[int] = None) -> None:
        if len(self._cache) >= self.max_size:
            await self._evict_lru()

        self._cache[key] = value
        self._access_times[key] = time.time()

    async def delete(self, key: str) -> None:
        self._cache.pop(key, None)
        self._access_times.pop(key, None)

    async def _evict_lru(self):
        """LRU淘汰策略"""
        if not self._access_times:
            return

        lru_key = min(self._access_times.keys(), key=lambda k: self._access_times[k])
        await self.delete(lru_key)

    async def clear(self):
        self._cache.clear()
        self._access_times.clear()

class MultiLevelCache:
    """多层缓存实现"""

    def __init__(self, l1_backend: CacheBackend, l2_backend: CacheBackend):
        self.l1_cache = l1_backend
        self.l2_cache = l2_backend
        self.l1_ttl = 300  # 5分钟
        self.l2_ttl = 3600  # 1小时

    async def get(self, key: str) -> Any:
        # 优先从L1缓存获取
        value = await self.l1_cache.get(key)
        if value is not None:
            return value

        # 从L2缓存获取
        value = await self.l2_cache.get(key)
        if value is not None:
            # 提升到L1缓存
            await self.l1_cache.set(key, value, self.l1_ttl)
            return value

        return None

    async def set(self, key: str, value: Any, expire: Optional[int] = None) -> None:
        # 并发设置到两层缓存
        tasks = [
            self.l1_cache.set(key, value, min(expire, self.l1_ttl)),
            self.l2_cache.set(key, value, expire)
        ]

        await asyncio.gather(*tasks, return_exceptions=True)

    async def delete(self, key: str) -> None:
        # 从两层缓存删除
        await asyncio.gather(
            self.l1_cache.delete(key),
            self.l2_cache.delete(key),
            return_exceptions=True
        )

    async def clear(self):
        await asyncio.gather(
            self.l1_cache.clear(),
            self.l2_cache.clear(),
```

```python
class VectorIndexOptimizer:
    """向量索引优化器"""

    def __init__(self, storage: SemanticStorage):
        self.storage = storage

    async def auto_optimize_indexes(self):
        """自动优化向量索引"""
        # 分析向量数据特征
        vector_stats = await self._analyze_vector_data()

        # 根据数据量选择最优索引类型
        optimal_index = await self._select_optimal_index_type(vector_stats)

        # 创建优化索引
        await self._create_optimized_index(optimal_index, vector_stats)

    async def _analyze_vector_data(self) -> Dict[str, Any]:
        """分析向量数据特征"""
        async with self.storage.get_connection() as conn:
            # 统计向量数量
            count_result = await conn.execute(
                "SELECT COUNT(*) as count FROM semantic_features"
            )
            vector_count = count_result.scalar()

            if vector_count == 0:
                return {"count": 0}

            # 分析向量分布
            embedding_stats = await conn.execute("""
                SELECT
                    AVG(embedding) as mean_vector,
                    STDDEV(embedding) as std_vector,
                    MIN(embedding <-> array_fill(0, ARRAY[1536])::vector) as min_distance,
                    MAX(embedding <-> array_fill(0, ARRAY[1536])::vector) as max_distance,
                    CORR(embedding, embedding) as self_correlation
                FROM semantic_features
                LIMIT 1000
            """)

            stats = embedding_result.first()

            return {
                "count": vector_count,
                "dimension": 1536,  # 根据配置获取
                "mean_norm": float(stats.mean_vector if stats.mean_vector else 0),
                "std_norm": float(stats.std_vector if stats.std_vector else 0),
                "self_correlation": float(stats.self_correlation if stats.self_correlation else 0),
                "min_distance": float(stats.min_distance if stats.min_distance else 0),
                "max_distance": float(stats.max_distance if stats.max_distance else 0)
            }

    async def _select_optimal_index_type(self, stats: Dict[str, Any]) -> str:
        """根据数据特征选择最优索引类型"""
        count = stats["count"]
        self_correlation = stats["self_correlation"]

        if count < 1000:
            return "exact"  # 小数据集使用精确搜索
        elif count < 100000:
            if self_correlation > 0.8:
                return "ivfflat"  # 高相关性使用IVFFlat
            else:
                return "hnsw"    # 低相关性使用HNSW
        else:
            return "hnsw"  # 大数据集使用HNSW

    async def _create_optimized_index(self, index_type: str, stats: Dict[str, Any]):
        """创建优化的索引"""
        if isinstance(self.storage, PostgreSQLVectorStore):
            await self._create_pgvector_index(index_type, stats)
        elif isinstance(self.storage, Neo4jVectorStore):
            await self._create_neo4j_index(index_type, stats)

    async def _create_pgvector_index(self, index_type: str, stats: Dict[str, Any]):
        """创建PostgreSQL pgvector索引"""
        async with self.storage._session_pool.acquire() as session:
            if index_type == "hnsw":
                # HNSW索引配置
                m = min(32, max(4, int(stats["count"] ** 0.25)))
                ef_construction = min(200, int(stats["count"] ** 0.5))

                await session.execute(f"""
                    CREATE INDEX CONCURRENTLY IF NOT EXISTS
                    idx_semantic_features_embedding_hnsw
                    ON semantic_features
                    USING hnsw (embedding vector_cosine_ops)
                    WITH (m = {m}, ef_construction = {ef_construction})
                """)

            elif index_type == "ivfflat":
                # IVFFlat索引配置
                lists = min(1000, max(32, int(stats["count"] / 100)))
                probes = min(100, int(lists * 0.1))

                await session.execute(f"""
                    CREATE INDEX CONCURRENTLY IF NOT EXISTS
                    idx_semantic_features_embedding_ivf
                    ON semantic_features
                    USING ivfflat (embedding vector_cosine_ops)
                    WITH (lists = {lists}, probes = {probes})
                """)

            await session.commit()

    async def create_statistics_maintenance_job(self):
        """创建统计维护任务"""
        # 定期更新表统计信息
        async with self.storage._session_pool.acquire() as session:
            await session.execute("ANALYZE semantic_features")
            await session.commit()

        # 更新向量索引统计
        await self._update_vector_index_stats()

    async def _update_vector_index_stats(self):
        """更新向量索引统计信息"""
        async with self.storage._session_pool.acquire() as conn:
            await conn.execute("""
                ANALYZE idx_semantic_features_embedding_hnsw
            """)
```

```python
class OptimizedVectorSearch:
    """优化的向量搜索实现"""

    def __init__(self, storage: VectorStore, cache_backend: CacheBackend):

                cached_result = await self.cache.get(cache_key) if enable_cache else None

                    return cached_result

                # 执行搜索
                result = await self.storage.vector_search(
                    query_vector=query_vector,
                    limit=limit,
                    min_similarity=min_similarity
                )

                # 缓存结果

                    await self.cache.set(cache_key, result, ttl=300)  # 5分钟

                return result

        # 并发执行搜索
        search_tasks = [
            single_search(vector, idx)
            for idx, vector in enumerate(query_vectors)
        ]

        search_results = await asyncio.gather(*search_tasks)
        return search_results

    async def adaptive_search(
        self,
        query: str,
        embedder: Embedder,
        initial_limit: int = 10,
        max_limit: int = 50
    ) -> List[Dict]:
        """自适应搜索：根据结果质量动态调整"""
        query_vector = await embedder.search_embed([query])
        query_vector = query_vector[0]

        # 第一阶段搜索
        initial_results = await self.storage.vector_search(
            query_vector=query_vector,
            limit=initial_limit,
            min_similarity=0.0  # 第一阶段使用较低的阈值
        )

        # 如果结果太少或质量不够，扩大搜索范围
        if len(initial_results) < 5:
            second_phase_results = await self.storage.vector_search(
                query_vector=query_vector,
                limit=max_limit,
                min_similarity=0.0
            )
            initial_results.extend(second_phase_results)

        elif self._check_result_quality(initial_results):
            # 结果质量不够好，重排序并扩展
            expanded_results = await self.storage.vector_search(
                query_vector=query_vector,
                limit=max_limit,
                min_similarity=0.0
            )

            # 合并并重排序
            all_results = self._merge_and_rerank_results(initial_results, expanded_results)
            return all_results[:max_limit]

        return initial_results

    def _check_result_quality(self, results: List[Dict]) -> bool:
        """检查搜索结果质量"""
        if len(results) < 3:
            return True

        # 计算相似度分布
        similarities = [r.get('similarity', 0.0) for r in results]
        avg_similarity = sum(similarities) / len(similarities)
        max_similarity = max(similarities)

        # 如果平均相似度太低或最高相似度不够高，认为质量不够
        return avg_similarity < 0.7 or max_similarity < 0.8

    def _merge_and_rerank_results(self, initial_results: List[Dict], expanded_results: List[Dict]) -> List[Dict]:
        """合并和重排序结果"""
        # 使用feature_id去重
        seen_ids = set()
        merged_results = {}

        # 保留初始结果
        for result in initial_results:
            feature_id = result.get('feature_id')
            if feature_id not in seen_ids:
                seen_ids.add(feature_id)
                merged_results[feature_id] = result
                merged_results[feature_id]['original_source'] = 'initial'

        # 添加扩展结果
        for result in expanded_results:
            feature_id = result.get('feature_id')
            if feature_id not in seen_ids:
                seen_ids.add(feature_id)
                merged_results[feature_id] = result
                merged_results[feature_id]['original_source'] = 'expanded'

        # 根据相似度和来源重新排序
        final_results = list(merged_results.values())
        final_results.sort(key=lambda x: (
            x.get('similarity', 0.0) * (1.0 if x['original_source'] == 'initial' else 0.9)
        ), reverse=True)

        return final_results
```

```python
import asyncio
import time
from typing import List, Callable, Any
from collections import deque

class AsyncTaskScheduler:
    """异步任务调度器"""

    def __init__(self, max_workers: int = 10, max_queue_size: int = 1000):
        self.max_workers = max_workers
        self.max_queue_size = max_queue_size
        self.task_queue = asyncio.Queue(maxsize=max_queue_size)
        self.workers: List[asyncio.Task] = []
        self.is_running = False
        self.task_stats = {
            'queued': 0,
            'completed': 0,
            'failed': 0,
            'average_latency': 0.0,
            'queue_size': 0
        }

    async def start(self):
        """启动任务调度器"""
        if self.is_running:
            return

        self.is_running = True

        # 启动工作协程
        for i in range(self.max_workers):
            worker = asyncio.create_task(self._worker(f"worker-{i}"))
            self.workers.append(worker)

        # 启动统计协程
        stats_task = asyncio.create_task(self._stats_updater())
        self.workers.append(stats_task)

    async def stop(self):
        """停止任务调度器"""
        self.is_running = False

        # 等待所有工作协程完成
        await asyncio.gather(*self.workers)

    async def submit_task(
        self,
        coro: Callable,
        *args,
        **kwargs
    ) -> asyncio.Future:
        """提交任务"""
        if not self.is_running:
            raise RuntimeError("Scheduler is not running")

        future = asyncio.Future()

        task_data = {
            'coro': coro,
            'args': args,
            'kwargs': kwargs,
            'future': future,
            'submit_time': time.time()
        }

        try:
            await self.task_queue.put_nowait(task_data)
            self.task_stats['queued'] += 1
        except asyncio.QueueFull:
            future.set_exception(QueueFullError("Task queue is full"))

        return future

    async def _worker(self, worker_name: str):
        """工作协程"""
        while self.is_running:
            try:
                task_data = await asyncio.wait_for(
                    self.task_queue.get(), timeout=1.0
                )

                # 执行任务
                start_time = time.time()
                try:
                    result = await task_data['coro'](
                        *task_data['args'],
                        **task_data['kwargs']
                    )

                    execution_time = time.time() - start_time

                    # 更新平均延迟
                    self._update_average_latency(execution_time)

                    task_data['future'].set_result(result)
                    self.task_stats['completed'] += 1

                except Exception as e:
                    task_data['future'].set_exception(e)
                    self.task_stats['failed'] += 1

            except asyncio.TimeoutError:
                # 超时继续下一个任务
                continue

    async def _stats_updater(self):
        """统计信息更新协程"""
        while self.is_running:
            self.task_stats['queue_size'] = self.task_queue.qsize()
            await asyncio.sleep(1)

    def _update_average_latency(self, latency: float):
        """更新平均延迟"""
        current_avg = self.task_stats['average_latency']
        completed = self.task_stats['completed']

        if completed == 0:
            self.task_stats['average_latency'] = latency
        else:
            # 指数移动平均
            alpha = 0.1
            self.task_stats['average_latency'] = (
                alpha * latency + (1 - alpha) * current_avg
            )

    def get_stats(self) -> Dict[str, Any]:
        """获取统计信息"""
        return self.task_stats.copy()
```

```python
        self,
        name: str,
        description: str,
        labels: Dict[str, str] = None
    ) -> Counter:
        """获取或创建计数器"""
        key = self._get_key(name, labels)

                description,
                labelnames=list(labels.keys()) if labels else []
            )

    def get_histogram(
        self,
        name: str,
        description: str,
        buckets: list[float] = None,
        labels: Dict[str, str] = None
    ) -> Histogram:
        """获取或创建直方图"""
        key = self._get_key(name, labels)
        if key not in self.histograms:
            self.histograms[key] = Histogram(
                name,
                description,
                labelnames=list(labels.keys()) if labels else [],
                buckets=buckets or self._default_buckets()
            )
        return self.histograms[key]

    def get_gauge(
        self,
        name: str,
        description: str,
        labels: Dict[str, str] = None
    ) -> Gauge:
        """获取或创建仪表盘"""
        key = self._get_key(name, labels)
        if key not in self.gauges:
            self.gauges[key] = Gauge(
                name,
                description,
                labelnames=list(labels.keys()) if labels else []
            )
        return self.gauges[key]

    def _get_key(self, name: str, labels: Dict[str, str] | None) -> str:
```

```python
class MemoryManager:
    """内存管理器"""

    def __init__(self, metrics_factory: PrometheusMetricsFactory):
        self.metrics = metrics_factory
        self.gc_stats = {}
        self.memory_pools = {}

    async def start_memory_monitoring(self):
        """启动内存监控"""
        while True:
            try:
                await self._collect_memory_stats()
                await self._trigger_gc_if_needed()
                await asyncio.sleep(30)  # 每30秒检查一次

            except Exception as e:
                self.metrics.get_counter("memory_monitor_errors").inc()
                await asyncio.sleep(30)

    async def _collect_memory_stats(self):
        """收集内存统计"""
        import psutil
        import gc

        # 进程内存使用
        process = psutil.Process()
        memory_info = process.memory_info()

        # 更新内存指标
        self.metrics.get_gauge("process_memory_rss_mb").set(
            memory_info.rss / 1024 / 1024
        )

        self.metrics.get_gauge("process_memory_vms_mb").set(
            memory_info.vms / 1024 / 1024
        )

        self.metrics.get_gauge("process_memory_percent").set(
            process.memory_percent()
        )

        # GC统计
        gc_stats = gc.get_stats()
        self.gc_stats['collections'] = gc_stats['collections']
        self.gc_stats['collected'] = gc_stats['collected']
        self.gc_stats['uncollectable'] = gc_stats['uncollectable']

        self.metrics.get_counter("gc_collections_total").inc(
            gc_stats['collections']
        )
        self.metrics.get_counter("gc_objects_collected").inc(
            gc_stats['collected']
        )

    async def _trigger_gc_if_needed(self):
        """如果需要则触发垃圾回收"""
        memory_percent = await self._get_memory_percent()

        if memory_percent > 80:  # 内存使用超过80%时触发GC
            import gc
            await asyncio.get_event_loop().run_in_executor(gc.collect)
            self.metrics.get_counter("forced_gc_triggers").inc()

    async def _get_memory_percent(self) -> float:
        """获取内存使用百分比"""
        import psutil
        return psutil.virtual_memory().percent
```

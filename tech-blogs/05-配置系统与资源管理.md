# 配置系统与资源管理

## 概述

MemMachine采用模块化的配置系统和统一的资源管理机制，为整个记忆系统提供灵活的配置能力和高效的生命周期管理。本文基于实际源代码深入分析其实现架构和使用方法。

## 配置系统架构

### 配置层次结构

```
Configuration (主配置)
├── EpisodicMemoryConfig (情景记忆配置)
├── SemanticMemoryConfig (语义记忆配置)
├── EmbedderConfig (嵌入器配置)
├── LanguageModelConfig (语言模型配置)
├── DatabaseConfig (数据库配置)
└── VectorStoreConfig (向量存储配置)
```

        """从环境变量获取API密钥"""
        if self.api_key:
            return self.api_key

        # 根据提供者查找环境变量
        env_vars = {
            EmbedderProvider.OPENAI: "OPENAI_API_KEY",
            EmbedderProvider.AWS_BEDROCK: "AWS_ACCESS_KEY_ID",
            EmbedderProvider.HUGGING_FACE: "HUGGINGFACE_API_KEY"
        }

        env_var = env_vars.get(self.provider)
        if env_var:
            import os
            return os.getenv(env_var)

        raise ValueError(f"API key not found for provider {self.provider}")

### 配置加载机制

```python
import os
from typing import Dict, Any
import yaml
from pathlib import Path

class ConfigurationManager:
    """配置管理器"""

    def __init__(self, config_path: Optional[str] = None):
        self.config_path = config_path
        self.config: Optional[Configuration] = None

    def load_configuration(self) -> Configuration:
        """加载配置"""
        if self.config_path and os.path.exists(self.config_path):
            # 从文件加载
            config_data = self._load_from_file(self.config_path)
        else:
            # 使用默认配置
            config_data = {}

        # 环境变量覆盖
        env_overrides = self._load_from_environment()
        self._merge_config(config_data, env_overrides)

        # 创建Configuration实例
        self.config = Configuration(**config_data)

        # 验证配置
        self._validate_configuration()

        return self.config

    def _load_from_file(self, config_path: str) -> Dict[str, Any]:
        """从文件加载配置"""
        config_file = Path(config_path)

        if config_file.suffix.lower() in ['.yml', '.yaml']:
            with open(config_file, 'r') as f:
                return yaml.safe_load(f) or {}
        elif config_file.suffix.lower() == '.json':
            import json
            with open(config_file, 'r') as f:
                return json.load(f)
        else:
            raise ValueError(f"Unsupported config file format: {config_file.suffix}")

    def _load_from_environment(self) -> Dict[str, Any]:
        """从环境变量加载配置"""
        env_config = {}

        # 环境变量前缀映射
        env_prefixes = {
            "MM_DB_": "database",
            "MM_EMBED_": "embedders.openai",
            "MM_LLM_": "language_models.openai",
            "MM_EPISODIC_": "episodic_memory",
            "MM_SEMANTIC_": "semantic_memory"
        }

        import os
        for env_var, config_path in env_prefixes.items():
            for key, value in os.environ.items():
                if key.startswith(env_var):
                    # 构建配置路径
                    config_keys = config_path.split('.')
                    config_key = key[len(env_var):].lower()
                    config_keys.append(config_key)

                    # 设置嵌套配置
                    current = env_config
                    for ck in config_keys[:-1]:
                        if ck not in current:
                            current[ck] = {}
                        current = current[ck]
                    current[config_keys[-1]] = self._convert_env_value(value)

        return env_config

    def _convert_env_value(self, value: str) -> Any:
        """转换环境变量值类型"""
        # 尝试转换为整数
        try:
            return int(value)
        except ValueError:
            pass

        # 尝试转换为浮点数
        try:
            return float(value)
        except ValueError:
            pass

        # 尝试转换为布尔值
        if value.lower() in ('true', 'false'):
            return value.lower() == 'true'

        # 尝试转换为JSON
        try:
            import json
            return json.loads(value)
        except json.JSONDecodeError:
            pass

        # 默认返回字符串
        return value

    def _merge_config(self, base_config: Dict[str, Any], override_config: Dict[str, Any]):
        """合并配置"""
        def deep_merge(base: dict, override: dict):
            for key, value in override.items():
                if key in base and isinstance(base[key], dict) and isinstance(value, dict):
                    deep_merge(base[key], value)
                else:
                    base[key] = value

        deep_merge(base_config, override_config)

    def _validate_configuration(self):
        """验证配置"""
        if not self.config:
            raise ValueError("Configuration not loaded")

        # 验证必要的嵌入器
        if not self.config.embedders:
            raise ValueError("At least one embedder must be configured")

        # 验证默认嵌入器
        if self.config.default_long_term_memory_embedder not in self.config.embedders:
            raise ValueError(f"Default embedder '{self.config.default_long_term_memory_embedder}' not found in embedders")
```

```python
import asyncio
from typing import Dict, Any, Optional
from pydantic import BaseModel, Field

class ResourceManagerImpl:
    """资源管理器实现"""

    def __init__(self, config: Configuration):
        self._config = config
        self._services: Dict[str, Any] = {}
        self._locks: Dict[str, asyncio.Lock] = {}
        self._initialized = False

    async def initialize(self):
        """初始化所有资源"""
        if self._initialized:
            return

        # 并发初始化各种服务
        init_tasks = [
            self._initialize_database(),
            self._initialize_embedders(),
            self._initialize_language_models(),
            self._initialize_vector_stores(),
            self._initialize_rerankers()
        ]

        await asyncio.gather(*init_tasks)
        self._initialized = True

    async def shutdown(self):
        """关闭所有资源"""
        if not self._initialized:
            return

        shutdown_tasks = []

        # 关闭各种服务
        if "episode_storage" in self._services:
            shutdown_tasks.append(self._services["episode_storage"].close())
        if "semantic_storage" in self._services:
            shutdown_tasks.append(self._services["semantic_storage"].close())

        await asyncio.gather(*shutdown_tasks, return_exceptions=True)
        self._services.clear()
        self._initialized = False

    async def get_episode_storage(self) -> EpisodeStorage:
        """获取对话存储服务"""
        if "episode_storage" not in self._services:
            async self._get_lock("episode_storage"):
                if "episode_storage" not in self._services:
                    storage = self._create_episode_storage()
                    await storage.startup()
                    self._services["episode_storage"] = storage

        return self._services["episode_storage"]

    async def get_semantic_storage(self) -> SemanticStorage:
        """获取语义存储服务"""
        if "semantic_storage" not in self._services:
            async self._get_lock("semantic_storage"):
                if "semantic_storage" not in self._services:
                    storage = self._create_semantic_storage()
                    await storage.startup()
                    self._services["semantic_storage"] = storage

        return self._services["semantic_storage"]

    async def get_embedder(self, name: str) -> Embedder:
        """获取嵌入器实例"""
        cache_key = f"embedder:{name}"

        if cache_key not in self._services:
            async self._get_lock(cache_key):
                if cache_key not in self._services:
                    embedder_config = self._config.embedders[name]
                    embedder = self._create_embedder(embedder_config)
                    self._services[cache_key] = embedder

        return self._services[cache_key]

    async def get_language_model(self, name: str) -> LanguageModel:
        """获取语言模型实例"""
        cache_key = f"language_model:{name}"

        if cache_key not in self._services:
            async self._get_lock(cache_key):
                if cache_key not in self._services:
                    lm_config = self._config.language_models[name]
                    language_model = self._create_language_model(lm_config)
                    self._services[cache_key] = language_model

        return self._services[cache_key]

    async def _get_lock(self, name: str) -> None:
        """获取或创建锁"""
        if name not in self._locks:
            self._locks[name] = asyncio.Lock()
        await self._locks[name].acquire()

    def _release_lock(self, name: str) -> None:
        """释放锁"""
        if name in self._locks:
            self._locks[name].release()

    def _create_episode_storage(self) -> EpisodeStorage:
        """创建对话存储服务"""
        database_config = self._config.database

        if database_config.type == "postgresql":
            from memmachine.common.episode_store.episode_sqlalchemy_store import (
                SQLAlchemyEpisodeStorage
            )
            return SQLAlchemyEpisodeStorage(
                connection_string=database_config.connection_string,
                pool_size=database_config.pool_size or 10,
                max_overflow=database_config.max_overflow or 20
            )
        elif database_config.type == "sqlite":
            from memmachine.common.episode_store.episode_sqlalchemy_store import (
                SQLAlchemyEpisodeStorage
            )
            return SQLAlchemyEpisodeStorage(
                database_url=database_config.sqlite_path,
                pool_size=database_config.pool_size or 5
            )
        else:
            raise ValueError(f"Unsupported database type: {database_config.type}")

    def _create_semantic_storage(self) -> SemanticStorage:
        """创建语义存储服务"""
        database_config = self._config.database

        if database_config.type == "postgresql":
            from memmachine.semantic_memory.storage.sqlalchemy_pgvector_semantic import (
                PostgreSQLSemanticStorage
            )
            from sqlalchemy.ext.asyncio import create_async_engine

            engine = create_async_engine(
                database_config.connection_string,
                pool_size=database_config.pool_size or 10,
                max_overflow=database_config.max_overflow or 20
            )

            return PostgreSQLSemanticStorage(
                session_factory=self._create_session_factory(engine),
                config=database_config
            )
        else:
            raise ValueError(f"PostgreSQL required for semantic storage, got: {database_config.type}")

    def _create_embedder(self, config: EmbedderConfig) -> Embedder:
        """创建嵌入器实例"""
        if config.provider == EmbedderProvider.OPENAI:
            from memmachine.common.embedder.openai_embedder import OpenAIEmbedder
            return OpenAIEmbedder(config)
        elif config.provider == EmbedderProvider.HUGGING_FACE:
            from memmachine.common.embedder.huggingface_embedder import HuggingFaceEmbedder
            return HuggingFaceEmbedder(config)
        elif config.provider == EmbedderProvider.AWS_BEDROCK:
            from memmachine.common.embedder.aws_bedrock_embedder import AWSBedrockEmbedder
            return AWSBedrockEmbedder(config)
        else:
            raise ValueError(f"Unsupported embedder provider: {config.provider}")

    def _create_language_model(self, config: LanguageModelConfig) -> LanguageModel:
        """创建语言模型实例"""
        if config.provider == "openai":
            from memmachine.common.language_model.openai_lm import OpenAILanguageModel
            return OpenAILanguageModel(config)
        elif config.provider == "anthropic":
            from memmachine.common.language_model.anthropic_lm import AnthropicLanguageModel
            return AnthropicLanguageModel(config)
        else:
            raise ValueError(f"Unsupported language model provider: {config.provider}")
```

### 服务注册表

```python
class ServiceRegistry:
    """服务注册表"""

    def __init__(self):
        self._services: Dict[str, ServiceDefinition] = {}
        self._instances: Dict[str, Any] = {}

    def register_service(self, service_def: ServiceDefinition):
        """注册服务定义"""
        self._services[service_def.name] = service_def

    async def get_service(self, name: str) -> Any:
        """获取服务实例"""
        if name not in self._instances:
            service_def = self._services.get(name)
            if not service_def:
                raise ValueError(f"Service '{name}' not registered")

            # 创建服务实例
            instance = await service_def.factory()
            self._instances[name] = instance

            # 如果服务支持启动，则启动它
            if hasattr(instance, 'startup') and callable(instance.startup):
                await instance.startup()

        return self._instances[name]

    async def shutdown_all(self):
        """关闭所有服务"""
        shutdown_tasks = []

        for name, instance in self._instances.items():
            if hasattr(instance, 'close') and callable(instance.close):
                shutdown_tasks.append(instance.close())
            elif hasattr(instance, 'shutdown') and callable(instance.shutdown):
                shutdown_tasks.append(instance.shutdown())

        await asyncio.gather(*shutdown_tasks, return_exceptions=True)
        self._instances.clear()

@dataclass
class ServiceDefinition:
    """服务定义"""
    name: str
    factory: Callable[[], Any]
    singleton: bool = True
    dependencies: List[str] = None
```

### 服务依赖注入

```python
class ServiceContainer:
    """服务容器，支持依赖注入"""

    def __init__(self):
        self._services: Dict[str, Any] = {}
        self._factories: Dict[str, Callable] = {}
        self._dependencies: Dict[str, List[str]] = {}

    def register(self, name: str, factory: Callable, dependencies: List[str] = None):
        """注册服务工厂"""
        self._factories[name] = factory
        self._dependencies[name] = dependencies or []

    def get(self, name: str) -> Any:
        """获取服务实例"""
        if name in self._services:
            return self._services[name]

        # 检查依赖
        dependencies = self._dependencies.get(name, [])
        for dep in dependencies:
            if dep not in self._services:
                self.get(dep)  # 递归创建依赖

        # 创建服务
        factory = self._factories[name]
        if name in dependencies:
            # 传递依赖给工厂
            dependency_instances = [self._services[dep] for dep in dependencies]
            instance = factory(*dependency_instances)
        else:
            instance = factory()

        self._services[name] = instance
        return instance
```

## 配置验证和错误处理

### 配置验证器

```python
from pydantic import ValidationError
import logging

class ConfigurationValidator:
    """配置验证器"""

    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def validate_configuration(self, config: Configuration) -> ValidationResult:
        """验证配置"""
        errors = []
        warnings = []

        # 验证嵌入器配置
        embedder_errors = self._validate_embedders(config)
        errors.extend(embedder_errors)

        # 验证数据库配置
        db_errors = self._validate_database(config)
        errors.extend(db_errors)

        # 验证语言模型配置
        lm_errors = self._validate_language_models(config)
        errors.extend(lm_errors)

        # 验证向量存储配置
        vector_errors = self._validate_vector_stores(config)
        errors.extend(vector_errors)

        # 验证记忆系统配置
        memory_errors = self._validate_memory_configs(config)
        errors.extend(memory_errors)

        return ValidationResult(
            is_valid=len(errors) == 0,
            errors=errors,
            warnings=warnings
        )

    def _validate_embedders(self, config: Configuration) -> List[str]:
        """验证嵌入器配置"""
        errors = []

        if not config.embedders:
            errors.append("No embedders configured")
            return errors

        # 检查默认嵌入器
        default_embedder = config.default_long_term_memory_embedder
        if default_embedder not in config.embedders:
            errors.append(f"Default embedder '{default_embedder}' not found")

        # 验证每个嵌入器配置
        for name, embedder_config in config.embedders.items():
            embedder_errors = self._validate_embedder_config(name, embedder_config)
            errors.extend(embedder_errors)

        return errors

    def _validate_embedder_config(self, name: str, config: EmbedderConfig) -> List[str]:
        """验证单个嵌入器配置"""
        errors = []

        # 检查模型名称
        if not config.model_name:
            errors.append(f"Embedder '{name}': model_name is required")

        # 检查API密钥
        if not config.api_key and config.provider not in ["local"]:
            try:
                config.get_env_api_key()
            except ValueError:
                errors.append(f"Embedder '{name}': API key not found")

        # 检查批处理大小
        if config.max_batch_size <= 0:
            errors.append(f"Embedder '{name}': max_batch_size must be positive")

        # 检查超时时间
        if config.timeout <= 0:
            errors.append(f"Embedder '{name}': timeout must be positive")

        return errors

    def _validate_database(self, config: Configuration) -> List[str]:
        """验证数据库配置"""
        errors = []

        db_config = config.database

        if not db_config.type:
            errors.append("Database type is required")

        if db_config.type == "postgresql":
            if not db_config.connection_string:
                errors.append("PostgreSQL connection_string is required")
        elif db_config.type == "sqlite":
            if not db_config.sqlite_path:
                errors.append("SQLite sqlite_path is required")
        else:
            errors.append(f"Unsupported database type: {db_config.type}")

        # 检查连接池配置
        if db_config.pool_size and db_config.pool_size <= 0:
            errors.append("Database pool_size must be positive")

        return errors

    def _validate_language_models(self, config: Configuration) -> List[str]:
        """验证语言模型配置"""
        errors = []

        for name, lm_config in config.language_models.items():
            lm_errors = self._validate_language_model_config(name, lm_config)
            errors.extend(lm_errors)

        return errors

    def _validate_language_model_config(self, name: str, config: LanguageModelConfig) -> List[str]:
        """验证单个语言模型配置"""
        errors = []

        if not config.model_name:
            errors.append(f"Language model '{name}': model_name is required")

        # 验证温度参数
        if config.temperature is not None and (config.temperature < 0 or config.temperature > 2):
            errors.append(f"Language model '{name}': temperature must be between 0 and 2")

        # 验证最大token数
        if config.max_tokens is not None and config.max_tokens <= 0:
            errors.append(f"Language model '{name}': max_tokens must be positive")

        return errors

@dataclass
class ValidationResult:
    """验证结果"""
    is_valid: bool
    errors: List[str]
    warnings: List[str]
```

    METRICS_HOST = "MM_METRICS_HOST"

    @classmethod
    def get_all_variables(cls) -> Dict[str, str]:
        """获取所有环境变量名"""
        return {
            name: getattr(cls, name)
            for name in dir(cls)
            if not name.startswith('_') and not callable(getattr(cls, name))
        }

    @classmethod
    def load_from_env(cls) -> Dict[str, Any]:
        """从环境变量加载配置"""
        config = {}

        import os
        for var_name in cls.get_all_variables().values():
            value = os.getenv(var_name)
            if value is not None:
                config_name = var_name.lower().replace('mm_', '')
                config[config_name] = value

        return config

# 语义记忆系统深度解析

## 概述

语义记忆系统（Semantic Memory System）是MemMachine的另一个核心组件，专注于从对话历史中提取、存储和检索语义特征。本文基于实际源代码深入分析其实现机制、核心算法和智能特性。

## 系统架构概览

### 核心组件关系

```
SemanticService (语义服务)
├── IngestionService (摄入服务)
├── SemanticStorage (存储抽象层)
│   ├── PostgreSQL+pgvector实现
│   └── Neo4j实现
├── ResourceRetriever (资源管理)
└── SemanticSessionManager (会话管理)
```

### 数据流架构

```
对话输入 → 摄入服务 → LLM特征提取 → 向量嵌入 → 存储层 → 语义检索
    ↓         ↓           ↓          ↓       ↓        ↓
 Episodes → 未处理队列 → 语义指令 → Embeddings → Database → Results
```

```python
from pydantic import BaseModel, InstanceOf

class SemanticCommandType(Enum):
    """语义内存操作类型"""
    ADD = "add"
    DELETE = "delete"

class SemanticCommand(BaseModel):
    """LLM发出的标准化语义特征变更指令"""
    command: SemanticCommandType
    feature: str
    tag: str
    value: str

class SemanticFeature(BaseModel):
    """语义记忆条目模型"""

    class Metadata(BaseModel):
        """语义特征的存储元数据"""
        citations: list[EpisodeIdT] | None = None  # 引用的对话条目
        id: FeatureIdT | None = None               # 特征ID
        other: dict[str, Any] | None = None        # 其他元数据

    set_id: SetIdT | None = None    # 特征集ID
    category: str                  # 特征类别
    tag: str                      # 特征标签
    feature_name: str             # 特征名称
    value: str                    # 特征值
    metadata: Metadata = Metadata()

    @staticmethod
    def group_features(
        features: list["SemanticFeature"],
    ) -> dict[tuple[str, str, str], list["SemanticFeature"]]:
        """按类别、标签、特征名称分组特征"""
        grouped_features: dict[tuple[str, str, str], list[SemanticFeature]] = {}

        for f in features:
            key = (f.category, f.tag, f.feature_name)
            if key not in grouped_features:
                grouped_features[key] = []
            grouped_features[key].append(f)

        return grouped_features

    @staticmethod
    def group_features_by_tag(
        features: list["SemanticFeature"],
    ) -> dict[tuple[str, str], list["SemanticFeature"]]:
        """按标签和特征名称分组特征"""
        grouped_features: dict[tuple[str, str], list["SemanticFeature"]] = {}

        for f in features:
            key = (f.tag, f.feature_name)
            if key not in grouped_features:
                grouped_features[key] = []
            grouped_features[key].append(f)

        return grouped_features
```

```python
import numpy as np
from pydantic import BaseModel, InstanceOf

class SemanticService:
    """用于摄入历史记录和服务语义功能的高级协调器"""

    class Params(BaseModel):
        """基础设施依赖和后台更新配置"""

        episode_storage: InstanceOf[EpisodeStorage]
        consolidation_threshold: int = 20                    # 整合阈值
        feature_update_interval_sec: float = 2.0           # 特征更新间隔
        feature_update_message_limit: int = 5              # 特征更新消息限制
        resource_retriever: InstanceOf[ResourceRetriever]  # 资源检索器
        debug_fail_loudly: bool = False                    # 调试失败报告

    def __init__(self, params: Params) -> None:
        """初始化语义内存服务"""
        self._semantic_storage = params.semantic_storage
        self._episode_storage = params.episode_storage
        self._background_ingestion_interval_sec = params.feature_update_interval_sec
        self._resource_retriever: ResourceRetriever = params.resource_retriever
        self._consolidation_threshold = params.consolidation_threshold
        self._feature_update_message_limit = params.feature_update_message_limit
        self._ingestion_task: Task | None = None
        self._is_shutting_down = False
        self._debug_fail_loudly = params.debug_fail_loudly

    async def start(self) -> None:
        """启动语义内存服务"""
        if self._ingestion_task is not None:
            return

        self._is_shutting_down = False
        self._ingestion_task = asyncio.create_task(self._background_ingestion_task())

    async def stop(self) -> None:
        """停止语义内存服务"""
        if self._ingestion_task is None:
            return

        self._is_shutting_down = True
        await self._ingestion_task
```

```python
async def search(
    self,
    set_ids: list[SetIdT],
    query: str,
    *,
    min_distance: float | None = None,
    limit: int | None = 30,
    load_citations: bool = False,

        filter_expr=filter_expr,
        page_size=limit,
        vector_search_opts=SemanticStorage.VectorSearchOpts(

            min_distance=min_distance,
        ),
        load_citations=load_citations,
    )
```

### 特征管理功能

```python
async def add_new_feature(
    self,
    *,
    set_id: SetIdT,
    category_name: str,
    feature: str,
    value: str,
    tag: str,
    metadata: dict[str, str] | None = None,
    citations: list[EpisodeIdT] | None = None,
) -> FeatureIdT:
    """添加新的语义特征"""
    resources = self._resource_retriever.get_resources(set_id)

    # 使用摄入嵌入器生成特征向量
    embedding = (await resources.embedder.ingest_embed([value]))[0]

    f_id = await self._semantic_storage.add_feature(
        set_id=set_id,
        category_name=category_name,
        feature=feature,
        value=value,
        tag=tag,
        metadata=metadata,
        embedding=np.array(embedding),
    )

    # 添加引用关系
    if citations is not None:
        await self._semantic_storage.add_citations(f_id, citations)

    return f_id

async def update_feature(
    self,
    feature_id: FeatureIdT,
    *,
    set_id: SetIdT | None = None,
    category_name: str | None = None,
    feature: str | None = None,
    value: str | None = None,
    tag: str | None = None,
    metadata: dict[str, str] | None = None,
) -> None:
    """更新语义特征"""
    if value is not None:
        if set_id is None:
            # 如果未提供set_id，从现有特征中推断
            original_feature = await self._semantic_storage.get_feature(feature_id)
            if original_feature is None or original_feature.set_id is None:
                raise ValueError(
                    "无法推断set_id，feature_id可能不正确。更新特征时需要set_id",
                )
            set_id = original_feature.set_id

        resources = self._resource_retriever.get_resources(set_id)

        # 为新值生成嵌入向量
        embedding = (await resources.embedder.ingest_embed([value]))[0]
    else:
        embedding = None

    await self._semantic_storage.update_feature(
        feature_id=feature_id,
        set_id=set_id,
        category_name=category_name,
        feature=feature,
        value=value,
        tag=tag,
        metadata=metadata,
        embedding=np.array(embedding),
    )
```

## 智能摄入服务

### IngestionService实现

基于`src/memmachine/semantic_memory/semantic_ingestion.py`的实际实现：

```python
import asyncio
import itertools
import logging
from itertools import chain
import numpy as np
from pydantic import BaseModel, InstanceOf, TypeAdapter

class IngestionService:
    """
    处理每个set_id的未摄入历史记录并更新语义特征
    该服务拉取待处理消息，调用LLM生成变更指令，应用结果变更，并可选择整合冗余记忆
    """

    class Params(BaseModel):
        """摄入工作流的依赖项和调优参数"""
        semantic_storage: InstanceOf[SemanticStorage]
        history_store: InstanceOf[EpisodeStorage]
        resource_retriever: InstanceOf[ResourceRetriever]
        consolidated_threshold: int = 20
        debug_fail_loudly: bool = False

    def __init__(self, params: Params) -> None:
        """初始化摄入服务"""
        self._semantic_storage = params.semantic_storage
        self._history_store = params.history_store
        self._resource_retriever = params.resource_retriever
        self._consolidation_threshold = params.consolidated_threshold
        self._debug_fail_loudly = params.debug_fail_loudly

    async def process_set_ids(self, set_ids: list[SetIdT]) -> None:
        """处理指定的特征集ID列表"""
        results = await asyncio.gather(
            *[self._process_single_set(set_id) for set_id in set_ids],
            return_exceptions=True,
        )

        errors = [r for r in results if isinstance(r, Exception)]
        if len(errors) > 0:
            raise ExceptionGroup("Failed to process set ids", errors)

    async def _process_single_set(self, set_id: str) -> None:
        """处理单个特征集"""
        resources = self._resource_retriever.get_resources(set_id)

        # 获取未处理的对话ID
        history_ids = await self._semantic_storage.get_history_messages(
            set_ids=[set_id],
            limit=50,
            is_ingested=False,
        )

        if len(resources.semantic_categories) == 0:
            # 如果没有语义类别，直接标记为已摄入
            await self._semantic_storage.mark_messages_ingested(
                set_id=set_id,
                history_ids=history_ids,
            )

        if len(history_ids) == 0:
            return

        # 获取原始对话消息
        raw_messages = await asyncio.gather(
            *[self._history_store.get_episode(h_id) for h_id in history_ids],
        )

        if len(raw_messages) != len([m for m in raw_messages if m is not None]):
            raise ValueError("Failed to retrieve messages. Invalid history_ids")

        messages = TypeAdapter(list[Episode]).validate_python(raw_messages)

        # 为每个语义类别处理消息
        async def process_semantic_type(
            semantic_category: InstanceOf[SemanticCategory],
        ) -> None:
            for message in messages:
                if message.uid is None:
                    raise ValueError(
                        "Message uid cannot be None when processing semantic types"
                    )

                # 使用LLM进行特征更新
                await llm_feature_update(
                    language_model=resources.language_model,
                    embedder=resources.embedder,
                    category=semantic_category,
                    message=message,
                    set_id=set_id,
                    storage=self._semantic_storage,
                )

        # 并发处理所有语义类别
        await asyncio.gather(
            *[process_semantic_type(cat) for cat in resources.semantic_categories]
        )

        # 标记消息为已摄入
        await self._semantic_storage.mark_messages_ingested(
            set_id=set_id,
            history_ids=history_ids,
        )

        # 检查是否需要整合
        await self._check_consolidation(set_id)
```

**摄入流程特性：**
- **批量处理**：一次处理多个消息
- **并发分类处理**：同时处理多个语义类别
- **LLM驱动**：使用大语言模型进行智能特征提取
- **状态管理**：跟踪消息的摄入状态
- **自动整合**：当特征数量超过阈值时自动整合

### 特征整合机制

```python
async def _check_consolidation(self, set_id: str) -> None:
    """检查并执行特征整合"""
    if self._consolidation_threshold <= 0:
        return

    for category in await self._semantic_storage.get_categories(set_id=set_id):
        if await self._semantic_storage.get_feature_count(
            set_id=set_id, category=category
        ) > self._consolidation_threshold:
            await self._consolidate_features(set_id, category)

async def _consolidate_features(self, set_id: str, category: str) -> None:
    """整合特征以减少冗余"""
    resources = self._resource_retriever.get_resources(set_id)

    # 获取指定类别的所有特征
    features = await self._semantic_storage.get_feature_set(
        set_id=set_id,
        category=category,
        page_size=None,  # 获取所有特征
    )

    # 使用LLM进行特征整合
    await llm_consolidate_features(
        language_model=resources.language_model,
        embedder=resources.embedder,
        features=features,
        storage=self._semantic_storage,
    )
```

## 后台摄入任务

```python
async def _background_ingestion_task(self) -> None:
    """后台摄入任务"""
    ingestion_service = IngestionService(
        params=IngestionService.Params(
            semantic_storage=self._semantic_storage,
            resource_retriever=self._resource_retriever,
            history_store=self._episode_storage,
            consolidation_threshold=self._consolidation_threshold,
            debug_fail_loudly=self._debug_fail_loudly,
        ),
    )

    while not self._is_shutting_down:
        try:
            # 获取需要处理的特征集
            dirty_sets = await self._semantic_storage.get_history_set_ids(
                min_uningested_messages=self._feature_update_message_limit,
            )

            if len(dirty_sets) == 0:
                await asyncio.sleep(self._background_ingestion_interval_sec)
                continue

            # 批量处理
            await ingestion_service.process_set_ids(dirty_sets)

        except Exception:
            if self._debug_fail_loudly:
                raise
            logger.exception("后台任务崩溃，正在重启")
```

### PostgreSQL+pgvector实现

基于`src/memmachine/semantic_memory/storage/sqlalchemy_pgvector_semantic.py`：

```python
import numpy as np
from sqlalchemy import select, text, and_

class PostgreSQLSemanticStorage(SemanticStorage):
    """基于PostgreSQL + pgvector的语义存储实现"""

    def __init__(self, session, config):
        self._session = session
        self._config = config

    async def add_feature(
        self,
        set_id: SetIdT,
        category_name: str,
        feature: str,
        value: str,
        tag: str,
        metadata: dict[str, str] | None,
        embedding: np.ndarray,
    ) -> FeatureIdT:
        """添加特征到PostgreSQL"""

        # 生成唯一ID
        feature_id = str(uuid.uuid4())

        # 插入特征记录
        query = text("""
        INSERT INTO semantic_features (
            feature_id, set_id, category_name, feature_name,
            value, tag, metadata, embedding, created_at
        ) VALUES (
            :feature_id, :set_id, :category_name, :feature_name,
            :value, :tag, :metadata, :embedding, NOW()
        )
        """)

        await self._session.execute(
            query,
            {
                "feature_id": feature_id,
                "set_id": set_id,
                "category_name": category_name,
                "feature_name": feature,
                "value": value,
                "tag": tag,
                "metadata": json.dumps(metadata) if metadata else None,
                "embedding": embedding.tolist()
            }
        )

        await self._session.commit()
        return feature_id

    async def get_feature_set(
        self,
        filter_expr: FilterExpr | None = None,
        page_size: int | None = None,
        page_num: int | None = None,
        vector_search_opts: VectorSearchOpts | None = None,
        load_citations: bool = False,
    ) -> list[SemanticFeature]:
        """从PostgreSQL获取特征集"""

        if vector_search_opts:
            return await self._vector_search(
                vector_search_opts, filter_expr, page_size, page_num, load_citations
            )
        else:
            return await self._filter_search(
                filter_expr, page_size, page_num, load_citations
            )

    async def _vector_search(
        self,
        vector_search_opts: VectorSearchOpts,
        filter_expr: FilterExpr | None,
        page_size: int | None,
        page_num: int | None,
        load_citations: bool,
    ) -> list[SemanticFeature]:
        """执行向量相似度搜索"""

        query = """
        SELECT *,
               embedding <-> :query_embedding as distance
        FROM semantic_features
        WHERE (:filter_expr IS NULL OR {filter_condition})
        ORDER BY embedding <-> :query_embedding
        LIMIT :limit
        """

        # 构建过滤条件
        filter_condition = ""
        if filter_expr:
            filter_condition = self._build_filter_condition(filter_expr)

        # 执行向量搜索
        result = await self._session.execute(
            text(query.format(filter_condition=filter_condition)),
            {
                "query_embedding": vector_search_opts.query_embedding.tolist(),
                "filter_expr": filter_expr,
                "limit": page_size or 30
            }
        )

        # 转换为SemanticFeature对象
        features = []
        for row in result:
            feature = self._row_to_feature(row)
            if load_citations:
                feature.metadata.citations = await self._get_feature_citations(row.feature_id)
            features.append(feature)

        return features
```

```python
async def llm_feature_update(
    language_model: InstanceOf[LanguageModel],
    embedder: InstanceOf[Embedder],
    category: InstanceOf[SemanticCategory],
    message: Episode,
    set_id: str,
    storage: InstanceOf[SemanticStorage],
) -> None:
    """使用LLM更新语义特征"""

    # 构建更新提示
    prompt = category.prompt.update_prompt.format(
        message=message.content,
        producer=message.producer_id,
        role=message.producer_role
    )

    # 调用LLM生成语义指令
    response = await language_model.generate_completion(
        system_prompt="你是一个专业的语义特征提取器。请分析对话并生成相应的语义特征操作。",
        user_prompt=prompt,
        temperature=0.1,
        max_tokens=1000
    )

    # 解析LLM响应
    try:
        commands = parse_llm_response_to_commands(response.content)
    except Exception as e:
        logger.error(f"Failed to parse LLM response: {e}")
        return

    # 执行语义指令
    for command in commands:
        try:
            if command.command == SemanticCommandType.ADD:
                await execute_add_command(
                    command, storage, embedder, set_id, category, message
                )
            elif command.command == SemanticCommandType.DELETE:
                await execute_delete_command(command, storage, set_id)
        except Exception as e:
            logger.error(f"Failed to execute command {command}: {e}")

async def execute_add_command(
    command: SemanticCommand,
    storage: InstanceOf[SemanticStorage],
    embedder: InstanceOf[Embedder],
    set_id: str,
    category: InstanceOf[SemanticCategory],
    message: Episode
) -> None:
    """执行添加特征命令"""

    # 生成嵌入向量
    embedding = (await embedder.ingest_embed([command.value]))[0]

    # 添加特征
    feature_id = await storage.add_feature(
        set_id=set_id,
        category_name=category.name,
        feature=command.feature,
        value=command.value,
        tag=command.tag,
        metadata={
            "producer_id": message.producer_id,
            "producer_role": message.producer_role,
            "message_id": message.uid,
            "timestamp": message.created_at.isoformat()
        },
        embedding=np.array(embedding),
    )

    # 添加引用关系
    if message.uid:
        await storage.add_citations(feature_id, [message.uid])

async def llm_consolidate_features(
    language_model: InstanceOf[LanguageModel],
    embedder: InstanceOf[Embedder],
    features: list[SemanticFeature],
    storage: InstanceOf[SemanticStorage],
) -> None:
    """使用LLM整合语义特征"""

    # 按类别分组特征
    grouped_features = SemanticFeature.group_features(features)

    for (category, tag, feature_name), feature_list in grouped_features.items():
        if len(feature_list) <= 5:  # 只整合数量较多的特征
            continue

        # 构建整合提示
        prompt = build_consolidation_prompt(feature_list)

        # 调用LLM
        response = await language_model.generate_completion(
            system_prompt="你是一个语义特征整合专家。请分析并整合相似的语义特征。",
            user_prompt=prompt,
            temperature=0.1,
            max_tokens=2000
        )

        # 解析并执行整合操作
        consolidation_ops = parse_consolidation_response(response.content)
        await execute_consolidation_ops(
            consolidation_ops, storage, embedder, category, tag, feature_name
        )
```

## 总结

基于实际源代码分析，MemMachine的语义记忆系统展现了以下核心特性：

1. **智能特征提取**：使用LLM从对话中自动提取语义特征
2. **向量嵌入管理**：支持多种嵌入器和向量搜索
3. **多存储后端**：支持PostgreSQL+pgvector和Neo4j等
4. **异步并发处理**：后台自动摄入和处理
5. **智能整合机制**：自动整合冗余特征
6. **会话隔离管理**：支持用户、角色、会话级别的隔离
7. **引用追踪**：维护特征与原始对话的关联关系
8. **LLM深度集成**：将大语言模型融入特征提取和管理流程

这种设计使得语义记忆系统能够智能地从对话中提取、组织和检索语义特征，为AI智能体提供深层次的理解和个性化能力。

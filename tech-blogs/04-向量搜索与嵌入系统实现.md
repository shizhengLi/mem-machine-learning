# 向量搜索与嵌入系统实现

## 概述

向量搜索与嵌入系统是MemMachine语义记忆的核心技术基础，负责将文本转换为高维向量，并基于向量相似度进行高效的语义检索。本文基于实际源代码深入分析其实现机制、算法优化和性能特性。

## 系统架构

### 核心组件关系

```
嵌入器层 (Embedder Layer)
├── OpenAI Embedder
├── Hugging Face Embedder
├── AWS Bedrock Embedder
└── Custom Embedder

向量存储层 (Vector Storage)
├── PostgreSQL + pgvector
├── Neo4j Vector Search
└── Custom Vector Store

搜索层 (Search Layer)
├── 向量相似度搜索
├── 混合搜索
└── 重排序器
```

```python
class OpenAIEmbedder:
    """OpenAI嵌入器实现"""

    def __init__(self, config: OpenAIEmbedderConfig):
        self.model_name = config.model_name
        self.client = OpenAI(api_key=config.api_key)
        self.dimension = self._get_model_dimension()

    def _get_model_dimension(self) -> int:
        """获取模型的嵌入维度"""
        dimension_map = {
            "text-embedding-ada-002": 1536,
            "text-embedding-3-small": 1536,
            "text-embedding-3-large": 3072,
        }
        return dimension_map.get(self.model_name, 1536)

    async def ingest_embed(
        self,
        texts: list[str],
    ) -> list[np.ndarray]:
        """为内容生成嵌入向量"""
        try:
            response = await self.client.embeddings.create(
                model=self.model_name,
                input=texts,
            )

            embeddings = []
            for embedding in response.data:
                vector = np.array(embedding.embedding, dtype=np.float32)
                # 归一化处理
                norm = np.linalg.norm(vector)
                if norm > 0:
                    vector = vector / norm
                embeddings.append(vector)

            return embeddings

        except Exception as e:
            logger.error(f"OpenAI embedding failed: {e}")
            raise EmbeddingError(f"Failed to embed texts: {e}")

    async def search_embed(
        self,
        queries: list[str],
    ) -> list[np.ndarray]:
        """为查询生成嵌入向量"""
        # 对于搜索，通常使用相同的嵌入逻辑
        return await self.ingest_embed(queries)
```

```python
from sentence_transformers import SentenceTransformer
import asyncio

class HuggingFaceEmbedder:
    """Hugging Face SentenceTransformer嵌入器"""

    def __init__(self, config: HuggingFaceEmbedderConfig):
        self.model_name = config.model_name
        self.device = config.device or "cpu"
        self.batch_size = config.batch_size or 32

        # 加载模型
        self.model = SentenceTransformer(
            self.model_name,
            device=self.device
        )

    async def ingest_embed(
        self,
        texts: list[str],
    ) -> list[np.ndarray]:
        """批量生成内容嵌入向量"""
        try:
            # 在线程池中运行CPU密集型操作
            loop = asyncio.get_event_loop()

            embeddings = await loop.run_in_executor(
                None,
                self._encode_batch,
                texts
            )

            return embeddings

        except Exception as e:
            logger.error(f"HuggingFace embedding failed: {e}")
            raise EmbeddingError(f"Failed to embed texts: {e}")

    def _encode_batch(self, texts: list[str]) -> list[np.ndarray]:
        """批量编码文本"""
        embeddings = self.model.encode(
            texts,
            batch_size=self.batch_size,
            convert_to_numpy=True,
            normalize_embeddings=True,  # 自动归一化
        )

        # 转换为列表格式
        return [emb.astype(np.float32) for emb in embeddings]

    async def search_embed(
        self,
        queries: list[str],
    ) -> list[np.ndarray]:
        """生成查询嵌入向量"""
        return await self.ingest_embed(queries)
```

```python
import numpy as np
from sqlalchemy import select, text, and_, func
from sqlalchemy.dialects.postgresql import ARRAY

class PostgreSQLVectorStore:
    """PostgreSQL + pgvector向量存储实现"""

    def __init__(self, session_pool, config):
        self._session_pool = session_pool
        self._config = config
        self._dimension = config.dimension or 1536

    async def create_table(self):
        """创建向量存储表"""
        async with self._session_pool.acquire() as session:
            await session.execute(text(f"""
                CREATE TABLE IF NOT EXISTS semantic_features (
                    feature_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
                    set_id VARCHAR(255) NOT NULL,
                    category_name VARCHAR(255) NOT NULL,
                    feature_name VARCHAR(255) NOT NULL,
                    value TEXT NOT NULL,
                    tag VARCHAR(255) NOT NULL,
                    metadata JSONB,
                    embedding VECTOR({self._dimension}) NOT NULL,
                    created_at TIMESTAMP DEFAULT NOW(),
                    updated_at TIMESTAMP DEFAULT NOW()
                );

                CREATE INDEX IF NOT EXISTS idx_semantic_features_set_id
                    ON semantic_features(set_id);

                CREATE INDEX IF NOT EXISTS idx_semantic_features_category
                    ON semantic_features(category_name);

                CREATE INDEX IF NOT EXISTS idx_semantic_features_tag
                    ON semantic_features(tag);

                CREATE INDEX IF NOT EXISTS idx_semantic_features_embedding
                    ON semantic_features USING ivfflat (embedding vector_cosine_ops);
            """))
            await session.commit()

    async def insert_vectors(
        self,
        vectors: list[dict]
    ) -> list[str]:
        """批量插入向量数据"""
        async with self._session_pool.acquire() as session:
            feature_ids = []

            for vector_data in vectors:
                feature_id = str(uuid.uuid4())
                feature_ids.append(feature_id)

                # 插入向量数据
                await session.execute(
                    text("""
                        INSERT INTO semantic_features (
                            feature_id, set_id, category_name,
                            feature_name, value, tag, metadata,
                            embedding, created_at
                        ) VALUES (
                            :feature_id, :set_id, :category_name,
                            :feature_name, :value, :tag, :metadata,
                            :embedding, NOW()
                        )
                    """),
                    {
                        "feature_id": feature_id,
                        "set_id": vector_data["set_id"],
                        "category_name": vector_data["category_name"],
                        "feature_name": vector_data["feature_name"],
                        "value": vector_data["value"],
                        "tag": vector_data["tag"],
                        "metadata": json.dumps(vector_data.get("metadata", {})),
                        "embedding": vector_data["embedding"].tolist()
                    }
                )

            await session.commit()
            return feature_ids

    async def vector_search(
        self,
        query_vector: np.ndarray,
        limit: int = 10,
        min_similarity: float = 0.0,
        set_id: str | None = None,
        category: str | None = None,
        tag: str | None = None
    ) -> list[dict]:
        """执行向量相似度搜索"""
        async with self._session_pool.acquire() as session:

            # 构建查询条件
            where_conditions = []
            params = {
                "query_embedding": query_vector.tolist(),
                "limit": limit,
                "min_similarity": min_similarity
            }

            if set_id:
                where_conditions.append("set_id = :set_id")
                params["set_id"] = set_id

            if category:
                where_conditions.append("category_name = :category")
                params["category"] = category

            if tag:
                where_conditions.append("tag = :tag")
                params["tag"] = tag

            where_clause = ""
            if where_conditions:
                where_clause = "WHERE " + " AND ".join(where_conditions)

            # 执行向量搜索
            query = text(f"""
                SELECT *,
                       1 - (embedding <=> :query_embedding) as similarity
                FROM semantic_features
                {where_clause}
                HAVING 1 - (embedding <=> :query_embedding) >= :min_similarity
                ORDER BY embedding <=> :query_embedding
                LIMIT :limit
            """)

            result = await session.execute(query, params)
            rows = result.fetchall()

            # 转换结果
            results = []
            for row in rows:
                results.append({
                    "feature_id": str(row.feature_id),
                    "set_id": row.set_id,
                    "category_name": row.category_name,
                    "feature_name": row.feature_name,
                    "value": row.value,
                    "tag": row.tag,
                    "metadata": json.loads(row.metadata) if row.metadata else {},
                    "similarity": float(row.similarity),
                    "created_at": row.created_at
                })

            return results

    async def create_hnsw_index(self):
        """创建HNSW索引用于更快的近似搜索"""
        async with self._session_pool.acquire() as session:
            await session.execute(text(f"""
                CREATE INDEX IF NOT EXISTS idx_semantic_features_embedding_hnsw
                ON semantic_features USING hnsw (embedding vector_cosine_ops)
                WITH (m = 16, ef_construction = 200);
            """))
            await session.commit()
```

```python
from neo4j import AsyncGraphDatabase
import numpy as np

class Neo4jVectorStore:
    """Neo4j向量存储实现"""

    def __init__(self, config: Neo4jConfig):
        self.driver = AsyncGraphDatabase.driver(
            config.uri,
            auth=(config.username, config.password)
        )
        self._dimension = config.dimension or 1536

    async def close(self):
        """关闭数据库连接"""
        await self.driver.close()

    async def create_constraints(self):
        """创建数据库约束"""
        async with self.driver.session() as session:
            await session.run("""
                CREATE CONSTRAINT feature_id_unique
                IF NOT EXISTS FOR (f:SemanticFeature)
                REQUIRE f.feature_id IS UNIQUE
            """)

            await session.run("""
                CREATE VECTOR INDEX semantic_feature_embedding
                IF NOT EXISTS FOR (f:SemanticFeature)
                ON (f.embedding)
                OPTIONS {indexType: 'exact', dimensions: %d}
            """ % self._dimension)

    async def insert_vector(
        self,
        feature_id: str,
        set_id: str,
        category: str,
        feature_name: str,
        value: str,
        tag: str,
        embedding: np.ndarray,
        metadata: dict | None = None
    ):
        """插入单个向量特征"""
        async with self.driver.session() as session:
            await session.run("""
                CREATE (f:SemanticFeature {
                    feature_id: $feature_id,
                    set_id: $set_id,
                    category: $category,
                    feature_name: $feature_name,
                    value: $value,
                    tag: $tag,
                    metadata: $metadata,
                    embedding: $embedding,
                    created_at: datetime()
                })
            """, {
                "feature_id": feature_id,
                "set_id": set_id,
                "category": category,
                "feature_name": feature_name,
                "value": value,
                "tag": tag,
                "metadata": json.dumps(metadata) if metadata else "{}",
                "embedding": embedding.tolist()
            })

    async def vector_search(
        self,
        query_vector: np.ndarray,
        limit: int = 10,
        min_similarity: float = 0.0,
        set_id: str | None = None
    ) -> list[dict]:
        """Neo4j向量相似度搜索"""
        async with self.driver.session() as session:
            # 构建WHERE条件
            where_clauses = ["1 - (vector.similarity(f.embedding, $query_embedding)) >= $min_similarity"]
            params = {
                "query_embedding": query_vector.tolist(),
                "min_similarity": min_similarity,
                "limit": limit
            }

            if set_id:
                where_clauses.append("f.set_id = $set_id")
                params["set_id"] = set_id

            where_clause = " AND ".join(where_clauses)

            result = await session.run(f"""
                CALL db.index.vector.queryNodes('semantic_feature_embedding', {{
                    index: 'semantic_feature_embedding',
                    queryEmbedding: $query_embedding,
                    topK: $limit
                }})
                YIELD node, score
                WHERE {where_clause}
                RETURN node {{
                    .feature_id, .set_id, .category, .feature_name,
                    .value, .tag, .metadata, .created_at
                }} as feature, score as similarity
                ORDER BY similarity DESC
            """, params)

            # 转换结果
            results = []
            for record in result:
                node = record["feature"]
                results.append({
                    "feature_id": node["feature_id"],
                    "set_id": node["set_id"],
                    "category": node["category"],
                    "feature_name": node["feature_name"],
                    "value": node["value"],
                    "tag": node["tag"],
                    "metadata": json.loads(node["metadata"]) if node["metadata"] else {},
                    "similarity": float(record["similarity"]),
                    "created_at": node["created_at"]
                })

            return results
```

## 向量搜索算法

### 相似度计算

```python
class VectorSimilarityCalculator:
    """向量相似度计算器"""

    @staticmethod
    def cosine_similarity(
        vector1: np.ndarray,
        vector2: np.ndarray
    ) -> float:
        """计算余弦相似度"""
        # 确保向量是一维的
        vector1 = vector1.flatten()
        vector2 = vector2.flatten()

        # 计算余弦相似度
        dot_product = np.dot(vector1, vector2)
        norm_product = np.linalg.norm(vector1) * np.linalg.norm(vector2)

        if norm_product == 0:
            return 0.0

        return float(dot_product / norm_product)

    @staticmethod
    def euclidean_distance(
        vector1: np.ndarray,
        vector2: np.ndarray
    ) -> float:
        """计算欧氏距离"""
        return float(np.linalg.norm(vector1 - vector2))

    @staticmethod
    def manhattan_distance(
        vector1: np.ndarray,
        vector2: np.ndarray
    ) -> float:
        """计算曼哈顿距离"""
        return float(np.sum(np.abs(vector1 - vector2)))

    @staticmethod
    def dot_product(
        vector1: np.ndarray,
        vector2: np.ndarray
    ) -> float:
        """计算点积"""
        return float(np.dot(vector1.flatten(), vector2.flatten()))
```

### 批量搜索优化

```python
class BatchVectorSearch:
    """批量向量搜索优化器"""

    def __init__(self, vector_store, batch_size: int = 100):
        self.vector_store = vector_store
        self.batch_size = batch_size

    async def batch_search(
        self,
        queries: list[str],
        embedder: Embedder,
        limit: int = 10,
        min_similarity: float = 0.0
    ) -> list[list[dict]]:
        """批量向量搜索"""
        # 生成查询向量
        query_vectors = await embedder.search_embed(queries)

        # 分批搜索
        results = []
        for i in range(0, len(query_vectors), self.batch_size):
            batch_vectors = query_vectors[i:i + self.batch_size]
            batch_queries = queries[i:i + self.batch_size]

            # 并发执行搜索
            search_tasks = [
                self._single_search_with_cache(
                    vector, query, limit, min_similarity
                )
                for vector, query in zip(batch_vectors, batch_queries)
            ]

            batch_results = await asyncio.gather(*search_tasks)
            results.extend(batch_results)

        return results

    async def _single_search_with_cache(
        self,
        query_vector: np.ndarray,
        query: str,
        limit: int,
        min_similarity: float
    ) -> list[dict]:
        """带缓存的单个搜索"""
        # 生成缓存键
        cache_key = f"{hash(query)}_{limit}_{min_similarity}"

        # 检查缓存
        cached_result = await self._get_from_cache(cache_key)
        if cached_result is not None:
            return cached_result

        # 执行搜索
        result = await self.vector_store.vector_search(
            query_vector=query_vector,
            limit=limit,
            min_similarity=min_similarity
        )

        # 缓存结果
        await self._store_to_cache(cache_key, result)

        return result
```

## 混合搜索系统

### 向量+关键词混合搜索

```python
class HybridSearchEngine:
    """混合搜索引擎：结合向量搜索和关键词搜索"""

    def __init__(self, vector_store, text_store, embedder):
        self.vector_store = vector_store
        self.text_store = text_store
        self.embedder = embedder

    async def hybrid_search(
        self,
        query: str,
        limit: int = 10,
        vector_weight: float = 0.7,
        keyword_weight: float = 0.3,
        filters: dict | None = None
    ) -> list[dict]:
        """执行混合搜索"""

        # 1. 向量相似度搜索
        query_vector = await self.embedder.search_embed([query])
        vector_results = await self.vector_store.vector_search(
            query_vector=query_vector[0],
            limit=limit,
            filters=filters
        )

        # 2. 关键词搜索
        keyword_results = await self.text_store.keyword_search(
            query=query,
            limit=limit,
            filters=filters
        )

        # 3. 结果融合
        merged_results = self._merge_search_results(
            vector_results, keyword_results,
            vector_weight, keyword_weight
        )

        # 4. 重排序
        final_results = self._rerank_results(merged_results, query)

        return final_results[:limit]

    def _merge_search_results(
        self,
        vector_results: list[dict],
        keyword_results: list[dict],
        vector_weight: float,
        keyword_weight: float
    ) -> list[dict]:
        """融合向量搜索和关键词搜索结果"""

        # 归一化权重
        total_weight = vector_weight + keyword_weight
        vector_weight /= total_weight
        keyword_weight /= total_weight

        # 创建结果映射
        result_map = {}

        # 处理向量结果
        for result in vector_results:
            feature_id = result["feature_id"]
            result_map[feature_id] = result.copy()
            result_map[feature_id]["vector_score"] = result.get("similarity", 0.0)
            result_map[feature_id]["keyword_score"] = 0.0
            result_map[feature_id]["final_score"] = (
                vector_weight * result["vector_score"]
            )

        # 处理关键词结果
        for result in keyword_results:
            feature_id = result["feature_id"]
            if feature_id in result_map:
                # 已存在，合并分数
                result_map[feature_id]["keyword_score"] = result.get("score", 0.0)
                result_map[feature_id]["final_score"] += (
                    keyword_weight * result["keyword_score"]
                )
            else:
                # 新结果
                result_copy = result.copy()
                result_copy["vector_score"] = 0.0
                result_copy["keyword_score"] = result.get("score", 0.0)
                result_copy["final_score"] = (
                    keyword_weight * result_copy["keyword_score"]
                )
                result_map[feature_id] = result_copy

        # 按最终分数排序
        merged_results = sorted(
            result_map.values(),
            key=lambda x: x["final_score"],
            reverse=True
        )

        return merged_results

    async def _rerank_results(
        self,
        results: list[dict],
        original_query: str
    ) -> list[dict]:
        """重排序搜索结果"""
        if not results:
            return results

        # 这里可以添加更复杂的重排序逻辑
        # 例如：使用交叉编码器、学习排序等
        return results
```

## 重排序系统

### CrossEncoder重排序

```python
from sentence_transformers import CrossEncoder
import asyncio

class CrossEncoderReranker:
    """基于CrossEncoder的重排序器"""

    def __init__(self, model_name: str = "ms-marco-MiniLM-L-6-v2"):
        self.model_name = model_name
        self.model = CrossEncoder(model_name)

    async def rerank(
        self,
        query: str,
        documents: list[str],
        top_k: int = 10
    ) -> list[dict]:
        """重排序文档"""
        if not documents:
            return []

        # 在线程池中运行CPU密集型操作
        loop = asyncio.get_event_loop()

        # 构建查询-文档对
        query_doc_pairs = [(query, doc) for doc in documents]

        # 获取重排序分数
        scores = await loop.run_in_executor(
            None,
            self.model.predict,
            query_doc_pairs
        )

        # 创建结果
        reranked_results = []
        for i, (doc, score) in enumerate(zip(documents, scores)):
            reranked_results.append({
                "document": doc,
                "original_index": i,
                "rerank_score": float(score)
            })

        # 按分数排序
        reranked_results.sort(key=lambda x: x["rerank_score"], reverse=True)

        return reranked_results[:top_k]
```

### 索引优化

```python
class IndexOptimizer:
    """向量索引优化器"""

    def __init__(self, vector_store):
        self.vector_store = vector_store

    async def optimize_indexes(self, force_rebuild: bool = False):
        """优化向量索引"""
        if isinstance(self.vector_store, PostgreSQLVectorStore):
            await self._optimize_pgvector_indexes(force_rebuild)
        elif isinstance(self.vector_store, Neo4jVectorStore):
            await self._optimize_neo4j_indexes(force_rebuild)

    async def _optimize_pgvector_indexes(self, force_rebuild: bool):
        """优化pgvector索引"""
        async with self.vector_store._session_pool.acquire() as session:
            # 统计向量数量
            result = await session.execute(
                text("SELECT COUNT(*) as count FROM semantic_features")
            )
            vector_count = result.scalar()

            if vector_count < 1000:
                logger.info("Vector count too small for index optimization")
                return

            # 根据数据量选择索引类型
            if vector_count > 100000:
                await session.execute(text("""
                    CREATE INDEX CONCURRENTLY IF NOT EXISTS
                    idx_semantic_features_embedding_hnsw
                    ON semantic_features USING hnsw (embedding vector_cosine_ops)
                    WITH (m = 16, ef_construction = 200)
                """))
            else:
                await session.execute(text("""
                    CREATE INDEX CONCURRENTLY IF NOT EXISTS
                    idx_semantic_features_embedding_ivf
                    ON semantic_features USING ivfflat (embedding vector_cosine_ops)
                    WITH (lists = 100, probes = 10)
                """))

            await session.commit()

    async def _optimize_neo4j_indexes(self, force_rebuild: bool):
        """优化Neo4j向量索引"""
        async with self.vector_store.driver.session() as session:
            # 统计节点数量
            result = await session.run("MATCH (f:SemanticFeature) RETURN count(f) as count")
            record = result.single()
            node_count = record["count"]

            if node_count < 1000:
                logger.info("Node count too small for index optimization")
                return

            # 如果需要重建索引
            if force_rebuild:
                await session.run("DROP INDEX semantic_feature_embedding IF EXISTS")

            # 创建或重建HNSW索引
            await session.run("""
```

### 缓存策略

```python
class VectorCache:
    """向量搜索缓存"""

    def __init__(self, cache_backend, ttl: int = 3600):
        self.cache = cache_backend
        self.ttl = ttl

    async def get_cached_search(
        self,
        query_hash: str,
        search_params: dict
    ) -> list[dict] | None:
        """获取缓存的搜索结果"""
        cache_key = f"vector_search:{query_hash}:{hash(str(search_params))}"

        cached_data = await self.cache.get(cache_key)
        if cached_data is not None:
            # 检查TTL
            if cached_data.get("timestamp", 0) + self.ttl > time.time():
                return cached_data["results"]
            else:
                # 过期，删除缓存
                await self.cache.delete(cache_key)

        return None

    async def cache_search_results(
        self,
        query_hash: str,
        search_params: dict,
        results: list[dict]
    ):
        """缓存搜索结果"""
        cache_key = f"vector_search:{query_hash}:{hash(str(search_params))}"

        cache_data = {
            "results": results,
            "timestamp": time.time(),
            "search_params": search_params
        }

        await self.cache.set(cache_key, cache_data, expire=self.ttl)
```

```python
class VectorSearchMetrics:
    """向量搜索性能监控"""

    def __init__(self, metrics_collector):
        self.metrics = metrics_collector

    async def record_search_latency(
        self,
        latency_ms: float,
        index_type: str,
        result_count: int
    ):
        """记录搜索延迟"""
        self.metrics.histogram(
            "vector_search_latency_ms",
            latency_ms,
            tags={
                "index_type": index_type,
                "result_count_bracket": self._get_result_count_bracket(result_count)
            }
        )

    async def record_index_size(self, index_type: str, size: int):
        """记录索引大小"""
        self.metrics.gauge(
            "vector_index_size",
            size,
            tags={"index_type": index_type}
        )

    async def record_similarity_distribution(
        self,
        similarities: list[float]
    ):
        """记录相似度分布"""
        for similarity in similarities:
            self.metrics.histogram(
                "vector_similarity_distribution",
                similarity,
                tags={"similarity_range": self._get_similarity_bracket(similarity)}
            )

    def _get_result_count_bracket(self, count: int) -> str:
        """获取结果数量区间"""
        if count <= 5:
            return "0-5"
        elif count <= 10:
            return "6-10"
        elif count <= 20:
            return "11-20"
        elif count <= 50:
            return "21-50"
        else:
            return "50+"

    def _get_similarity_bracket(self, similarity: float) -> str:
        """获取相似度区间"""
        if similarity >= 0.9:
            return "0.9-1.0"
        elif similarity >= 0.8:
            return "0.8-0.9"
        elif similarity >= 0.7:
            return "0.7-0.8"
        elif similarity >= 0.5:
            return "0.5-0.7"
        else:
            return "0.0-0.5"
```
